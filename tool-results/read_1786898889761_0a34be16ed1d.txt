     1→"use client";
     2→
     3→import { useEffect, useState, useCallback } from "react";
     4→import { motion } from "framer-motion";
     5→import { Card, CardContent, CardDescription, CardHeader, CardTitle } from "@/components/ui/card";
     6→import { Button } from "@/components/ui/button";
     7→import { Input } from "@/components/ui/input";
     8→import { Label } from "@/components/ui/label";
     9→import { Textarea } from "@/components/ui/textarea";
    10→import { Badge } from "@/components/ui/badge";
    11→import { Separator } from "@/components/ui/separator";
    12→import {
    13→  Dialog,
    14→  DialogContent,
    15→} from "@/components/ui/dialog";
    16→import {
    17→  Tabs,
    18→  TabsContent,
    19→  TabsList,
    20→  TabsTrigger,
    21→} from "@/components/ui/tabs";
    22→import {
    23→  AlertDialog,
    24→  AlertDialogAction,
    25→  AlertDialogCancel,
    26→  AlertDialogContent,
    27→  AlertDialogDescription,
    28→  AlertDialogFooter,
    29→  AlertDialogHeader,
    30→  AlertDialogTitle,
    31→} from "@/components/ui/alert-dialog";
    32→import { ElectionShell } from "@/components/dashboard/election-shell";
    33→import { ErrorState } from "@/components/dashboard/dashboard-skeleton";
    34→import { apiFetch } from "@/lib/api-fetch";
    35→import { toast } from "sonner";
    36→import { cn, formatCurrency, formatNumber } from "@/lib/utils";
    37→import {
    38→  Wallet,
    39→  CreditCard,
    40→  Handshake,
    41→  CheckCircle2,
    42→  Loader2,
    43→  Receipt,
    44→  Users,
    45→  Tag,
    46→  CircleDollarSign,
    47→  Sparkles,
    48→  ShieldCheck,
    49→  Copy,
    50→  ArrowRight,
    51→  Link as LinkIcon,
    52→} from "lucide-react";
    53→import type { CommercialActivationDTO } from "@/components/dashboard/types";
    54→
    55→interface ActivationResponse {
    56→  activation: CommercialActivationDTO;
    57→}
    58→
    59→const STATUS_LABELS: Record<string, { label: string; tone: string }> = {
    60→  UNPRICED: { label: "Unpriced", tone: "bg-muted text-muted-foreground border-border" },
    61→  PAYMENT_REQUIRED: { label: "Payment required", tone: "bg-amber-100 text-amber-700 border-amber-200 dark:bg-amber-950/40 dark:text-amber-300 dark:border-amber-900" },
    62→  PAYMENT_PENDING: { label: "Payment pending", tone: "bg-amber-100 text-amber-700 border-amber-200 dark:bg-amber-950/40 dark:text-amber-300 dark:border-amber-900" },
    63→  PAYMENT_VERIFIED: { label: "Payment verified", tone: "bg-emerald-100 text-emerald-700 border-emerald-300 dark:bg-emerald-950/50 dark:text-emerald-300 dark:border-emerald-800" },
    64→  NEGOTIATION_REQUESTED: { label: "Negotiation requested", tone: "bg-sky-100 text-sky-700 border-sky-200 dark:bg-sky-950/40 dark:text-sky-300 dark:border-sky-900" },
    65→  NEGOTIATION_IN_PROGRESS: { label: "Negotiation in progress", tone: "bg-sky-100 text-sky-700 border-sky-200 dark:bg-sky-950/40 dark:text-sky-300 dark:border-sky-900" },
    66→  NEGOTIATED_SETTLEMENT_PENDING: { label: "Settlement pending", tone: "bg-violet-100 text-violet-700 border-violet-200 dark:bg-violet-950/40 dark:text-violet-300 dark:border-violet-900" },
    67→  MANUALLY_APPROVED: { label: "Manually approved", tone: "bg-emerald-100 text-emerald-700 border-emerald-300 dark:bg-emerald-950/50 dark:text-emerald-300 dark:border-emerald-800" },
    68→  DECLINED: { label: "Declined", tone: "bg-destructive/10 text-destructive border-destructive/30" },
    69→};
    70→
    71→interface PayResponse {
    72→  payment: { reference: string; amount: number; status: string };
    73→}
    74→
    75→export default function ActivatePage({
    76→  params,
    77→}: {
    78→  params: Promise<{ id: string }>;
    79→}) {
    80→  const [electionId, setElectionId] = useState<string | null>(null);
    81→  const [activation, setActivation] = useState<CommercialActivationDTO | null>(null);
    82→  const [loading, setLoading] = useState(true);
    83→  const [error, setError] = useState<string | null>(null);
    84→  const [payOpen, setPayOpen] = useState(false);
    85→  const [paying, setPaying] = useState(false);
    86→  const [paymentResult, setPaymentResult] = useState<{ reference: string; amount: number } | null>(null);
    87→
    88→  // Negotiation form state
    89→  const [contactName, setContactName] = useState("");
    90→  const [contactEmail, setContactEmail] = useState("");
    91→  const [contactPhone, setContactPhone] = useState("");
    92→  const [preferredResponseChannel, setPreferredResponseChannel] = useState<"EMAIL" | "WHATSAPP" | "PHONE">("EMAIL");
    93→  const [message, setMessage] = useState("");
    94→  const [proposedAmount, setProposedAmount] = useState("");
    95→  const [submitting, setSubmitting] = useState(false);
    96→  const [negotiationSent, setNegotiationSent] = useState(false);
    97→  const [showCelebration, setShowCelebration] = useState(false);
    98→  const [orgSlug, setOrgSlug] = useState<string | null>(null);
    99→  const [orgName, setOrgName] = useState<string | null>(null);
   100→  const [hasAutoOpened, setHasAutoOpened] = useState(false);
   101→
   102→  useEffect(() => {
   103→    (async () => {
   104→      const { id } = await params;
   105→      setElectionId(id);
   106→    })();
   107→  }, [params]);
   108→
   109→  const load = useCallback(async () => {
   110→    if (!electionId) return;
   111→    setLoading(true);
   112→    setError(null);
   113→    const res = await apiFetch<ActivationResponse & { organization?: { slug: string; name: string; domain?: string | null; domainStatus?: string | null } | null }>(
   114→      `/api/elections/${electionId}/activation`
   115→    );
   116→    setLoading(false);
   117→    if (!res.success || !res.data) {
   118→      setError(res.error?.message ?? "Could not load activation");
   119→      return;
   120→    }
   121→    setActivation(res.data.activation);
   122→    if (res.data.organization) {
   123→      setOrgSlug(res.data.organization.slug);
   124→      setOrgName(res.data.organization.name);
   125→    }
   126→    // Auto-open the celebration when activation has been approved
   127→    // (either via payment or via negotiation approval by the platform admin)
   128→    // and we haven't shown it yet for this session.
   129→    const isActivated =
   130→      res.data.activation.status === "MANUALLY_APPROVED" ||
   131→      res.data.activation.status === "PAYMENT_VERIFIED";
   132→    if (isActivated && !hasAutoOpened) {
   133→      setShowCelebration(true);
   134→      setHasAutoOpened(true);
   135→    }
   136→  }, [electionId, hasAutoOpened]);
   137→
   138→  useEffect(() => {
   139→    let cancelled = false;
   140→    (async () => {
   141→      if (cancelled) return;
   142→      await load();
   143→    })();
   144→    return () => {
   145→      cancelled = true;
   146→    };
   147→  }, [load]);
   148→
   149→  async function pay() {
   150→    if (!electionId) return;
   151→    setPaying(true);
   152→    const res = await apiFetch<PayResponse>(
   153→      `/api/elections/${electionId}/activation/pay`,
   154→      { method: "POST" }
   155→    );
   156→    setPaying(false);
   157→    setPayOpen(false);
   158→    if (!res.success || !res.data?.payment) {
   159→      toast.error("Payment failed", { description: res.error?.message });
   160→      return;
   161→    }
   162→    setPaymentResult({
   163→      reference: res.data.payment.reference,
   164→      amount: res.data.payment.amount,
   165→    });
   166→    toast.success("Payment received!", {
   167→      description: `Reference: ${res.data.payment.reference}`,
   168→    });
   169→    setShowCelebration(true);
   170→    load();
   171→  }
   172→
   173→  async function submitNegotiation(e: React.FormEvent) {
   174→    e.preventDefault();
   175→    if (!electionId) return;
   176→    if (contactName.trim().length < 2) {
   177→      toast.error("Contact name is too short");
   178→      return;
   179→    }
   180→    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(contactEmail)) {
   181→      toast.error("Enter a valid email");
   182→      return;
   183→    }
   184→    setSubmitting(true);
   185→    const body = {
   186→      contactName: contactName.trim(),
   187→      contactEmail: contactEmail.trim(),
   188→      contactPhone: contactPhone.trim() || null,
   189→      preferredResponseChannel,
   190→      message: message.trim() || null,
   191→      proposedAmount: proposedAmount ? Number(proposedAmount) : null,
   192→    };
   193→    const res = await apiFetch<{ negotiation: unknown }>(
   194→      `/api/elections/${electionId}/activation/negotiate`,
   195→      { method: "POST", body: JSON.stringify(body) }
   196→    );
   197→    setSubmitting(false);
   198→    if (!res.success) {
   199→      toast.error("Could not submit request", { description: res.error?.message });
   200→      return;
   201→    }
   202→    toast.success("Negotiation request sent", {
   203→      description: "Our team will reach out within one business day.",
   204→    });
   205→    setNegotiationSent(true);
   206→    load();
   207→  }
   208→
   209→  const isVerified = activation?.status === "PAYMENT_VERIFIED" || activation?.status === "MANUALLY_APPROVED";
   210→
   211→  return (
   212→    <ElectionShell electionId={electionId ?? ""} activeTab="activate">
   213→      <div className="space-y-4">
   214→        <div className="flex flex-wrap items-center justify-between gap-3">
   215→          <div>
   216→            <h2 className="text-lg font-semibold tracking-tight">Activation</h2>
   217→            <p className="text-sm text-muted-foreground">
   218→              Paid-plan elections require activation before going LIVE. Choose to pay the standard rate or request a negotiation.
   219→            </p>
   220→          </div>
   221→          {activation && (
   222→            <Badge
   223→              variant="outline"
   224→              className={cn("text-xs", STATUS_LABELS[activation.status]?.tone ?? STATUS_LABELS.UNPRICED.tone)}
   225→            >
   226→              {STATUS_LABELS[activation.status]?.label ?? activation.status.replace(/_/g, " ").toLowerCase()}
   227→            </Badge>
   228→          )}
   229→        </div>
   230→
   231→        {loading ? (
   232→          <div className="grid gap-4 sm:grid-cols-3">
   233→            {Array.from({ length: 3 }).map((_, i) => (
   234→              <div key={i} className="h-32 animate-pulse rounded-xl bg-muted" />
   235→            ))}
   236→          </div>
   237→        ) : error ? (
   238→          <ErrorState message={error} onRetry={load} />
   239→        ) : activation ? (
   240→          <>
   241→            {/* Already activated state */}
   242→            {isVerified && (
   243→              <Card className="overflow-hidden border-primary/30 bg-primary/5">
   244→                <CardContent className="flex flex-wrap items-center gap-3 p-4">
   245→                  <div className="grid h-10 w-10 shrink-0 place-items-center rounded-full bg-emerald-100 text-emerald-600 dark:bg-emerald-950/40 dark:text-emerald-300">
   246→                    <CheckCircle2 className="h-5 w-5" />
   247→                  </div>
   248→                  <div className="min-w-0 flex-1">
   249→                    <p className="text-sm font-semibold">Election activated</p>
   250→                    <p className="text-xs text-muted-foreground">
   251→                      This election is cleared to go LIVE. Head back to Overview and transition to LIVE.
   252→                    </p>
   253→                  </div>
   254→                  {activation.activatedAt && (
   255→                    <Badge variant="outline" className="text-[10px]">
   256→                      <ShieldCheck className="h-3 w-3" />
   257→                      Activated {new Date(activation.activatedAt).toLocaleDateString()}
   258→                    </Badge>
   259→                  )}
   260→                </CardContent>
   261→              </Card>
   262→            )}
   263→
   264→            <div className="grid gap-4 lg:grid-cols-3">
   265→              {/* Pricing summary */}
   266→              <Card className="lg:col-span-1">
   267→                <CardHeader className="border-b pb-4">
   268→                  <CardTitle className="flex items-center gap-2 text-base">
   269→                    <Wallet className="h-4 w-4 text-primary" />
   270→                    Pricing summary
   271→                  </CardTitle>
   272→                  <CardDescription>Per-voter rate · snapshot at quote time.</CardDescription>
   273→                </CardHeader>
   274→                <CardContent className="space-y-3 pt-4">
   275→                  <Row
   276→                    icon={Users}
   277→                    label="Eligible voters"
   278→                    value={formatNumber(activation.voterCount)}
   279→                  />
   280→                  <Row
   281→                    icon={Tag}
   282→                    label="Applicable rate"
   283→                    value={`${formatCurrency(activation.applicableRate, activation.currency)} / voter`}
   284→                  />
   285→                  <Row
   286→                    icon={Sparkles}
   287→                    label="Pricing rule"
   288→                    value={
   289→                      <Badge variant="outline" className="text-[10px]">
   290→                        {activation.pricingRule}
   291→                      </Badge>
   292→                    }
   293→                  />
   294→                  <Separator />
   295→                  <Row
   296→                    icon={CircleDollarSign}
   297→                    label="Total amount"
   298→                    value={
   299→                      <span className="text-lg font-bold tabular-nums">
   300→                        {formatCurrency(activation.calculatedAmount, activation.currency)}
   301→                      </span>
   302→                    }
   303→                    highlight
   304→                  />
   305→                  <p className="text-[11px] text-muted-foreground">
   306→                    Bulk pricing applies above 2,000 voters (₦300/voter instead of ₦400/voter).
   307→                  </p>
   308→                </CardContent>
   309→              </Card>
   310→
   311→              {/* Action panel */}
   312→              <Card className="lg:col-span-2">
   313→                <CardHeader className="border-b pb-4">
   314→                  <CardTitle className="text-base">Choose activation path</CardTitle>
   315→                  <CardDescription>
   316→                    {isVerified
   317→                      ? "This election is already activated — no further action needed."
   318→                      : "Pay now or request a custom quote. Both paths unlock LIVE."}
   319→                  </CardDescription>
   320→                </CardHeader>
   321→                <CardContent className="pt-4">
   322→                  {paymentResult ? (
   323→                    <motion.div
   324→                      initial={{ opacity: 0, y: 6 }}
   325→                      animate={{ opacity: 1, y: 0 }}
   326→                      className="flex flex-col items-center gap-3 py-6 text-center"
   327→                    >
   328→                      <div className="grid h-14 w-14 place-items-center rounded-full bg-emerald-100 text-emerald-600 dark:bg-emerald-950/40 dark:text-emerald-300">
   329→                        <CheckCircle2 className="h-7 w-7" />
   330→                      </div>
   331→                      <div>
   332→                        <p className="text-lg font-semibold">Payment confirmed</p>
   333→                        <p className="text-sm text-muted-foreground">
   334→                          Your election is now activated and ready to go LIVE.
   335→                        </p>
   336→                      </div>
   337→                      <div className="mt-2 flex items-center gap-2 rounded-lg border bg-muted/40 px-4 py-2 text-sm">
   338→                        <Receipt className="h-4 w-4 text-muted-foreground" />
   339→                        <span className="font-mono text-xs">{paymentResult.reference}</span>
   340→                      </div>
   341→                      <Button variant="outline" size="sm" onClick={() => setPaymentResult(null)}>
   342→                        Dismiss
   343→                      </Button>
   344→                    </motion.div>
   345→                  ) : negotiationSent ? (
   346→                    <motion.div
   347→                      initial={{ opacity: 0, y: 6 }}
   348→                      animate={{ opacity: 1, y: 0 }}
   349→                      className="flex flex-col items-center gap-3 py-6 text-center"
   350→                    >
   351→                      <div className="grid h-14 w-14 place-items-center rounded-full bg-sky-100 text-sky-600 dark:bg-sky-950/40 dark:text-sky-300">
   352→                        <Handshake className="h-7 w-7" />
   353→                      </div>
   354→                      <div>
   355→                        <p className="text-lg font-semibold">Negotiation request sent</p>
   356→                        <p className="text-sm text-muted-foreground">
   357→                          Our team will reach out to <span className="font-medium">{contactEmail}</span> within one business day.
   358→                        </p>
   359→                      </div>
   360→                      <Button
   361→                        variant="outline"
   362→                        size="sm"
   363→                        onClick={() => {
   364→                          setNegotiationSent(false);
   365→                          setContactName("");
   366→                          setContactEmail("");
   367→                          setContactPhone("");
   368→                          setMessage("");
   369→                          setProposedAmount("");
   370→                        }}
   371→                      >
   372→                        Submit another
   373→                      </Button>
   374→                    </motion.div>
   375→                  ) : isVerified ? (
   376→                    <div className="flex flex-col items-center gap-3 py-6 text-center text-sm text-muted-foreground">
   377→                      <ShieldCheck className="h-10 w-10 text-primary" />
   378→                      No further activation action required.
   379→                    </div>
   380→                  ) : (
   381→                    <Tabs defaultValue="pay">
   382→                      <TabsList className="grid w-full grid-cols-2">
   383→                        <TabsTrigger value="pay">
   384→                          <CreditCard className="h-3.5 w-3.5" />
   385→                          Pay &amp; activate
   386→                        </TabsTrigger>
   387→                        <TabsTrigger value="negotiate">
   388→                          <Handshake className="h-3.5 w-3.5" />
   389→                          Request negotiation
   390→                        </TabsTrigger>
   391→                      </TabsList>
   392→
   393→                      <TabsContent value="pay" className="mt-4 space-y-4">
   394→                        <div className="rounded-lg border bg-muted/30 p-4">
   395→                          <p className="text-xs text-muted-foreground">
   396→                            You will be charged
   397→                          </p>
   398→                          <p className="mt-1 text-2xl font-bold tabular-nums">
   399→                            {formatCurrency(activation.calculatedAmount, activation.currency)}
   400→                          </p>
   401→                          <p className="mt-1 text-xs text-muted-foreground">
   402→                            for {formatNumber(activation.voterCount)} eligible voters at{" "}
   403→                            {formatCurrency(activation.applicableRate, activation.currency)}/voter
   404→                          </p>
   405→                        </div>
   406→                        <Button
   407→                          className="w-full"
   408→                          size="lg"
   409→                          onClick={() => setPayOpen(true)}
   410→                          disabled={!activation || activation.voterCount === 0}
   411→                        >
   412→                          <CreditCard className="h-4 w-4" />
   413→                          Pay {formatCurrency(activation.calculatedAmount, activation.currency)} &amp; activate
   414→                        </Button>
   415→                        {activation.voterCount === 0 && (
   416→                          <p className="text-center text-xs text-muted-foreground">
   417→                            Add eligible voters before activating — pricing is per voter.
   418→                          </p>
   419→                        )}
   420→                        <p className="text-center text-[11px] text-muted-foreground">
   421→                          <ShieldCheck className="mr-1 inline h-3 w-3" />
   422→                          Payments are processed securely. A receipt with a verifiable reference is issued instantly.
   423→                        </p>
   424→                      </TabsContent>
   425→
   426→                      <TabsContent value="negotiate" className="mt-4">
   427→                        <form onSubmit={submitNegotiation} className="grid gap-4">
   428→                          <div className="grid gap-4 sm:grid-cols-2">
   429→                            <div className="grid gap-1.5">
   430→                              <Label htmlFor="neg-name">Contact name *</Label>
   431→                              <Input
   432→                                id="neg-name"
   433→                                value={contactName}
   434→                                onChange={(e) => setContactName(e.target.value)}
   435→                                placeholder="e.g. Adaeze Okafor"
   436→                                required
   437→                              />
   438→                            </div>
   439→                            <div className="grid gap-1.5">
   440→                              <Label htmlFor="neg-email">Contact email *</Label>
   441→                              <Input
   442→                                id="neg-email"
   443→                                type="email"
   444→                                value={contactEmail}
   445→                                onChange={(e) => setContactEmail(e.target.value)}
   446→                                placeholder="finance@unizik.edu.ng"
   447→                                required
   448→                              />
   449→                            </div>
   450→                          </div>
   451→                          <div className="grid gap-4 sm:grid-cols-2">
   452→                            <div className="grid gap-1.5">
   453→                              <Label htmlFor="neg-phone">Phone</Label>
   454→                              <Input
   455→                                id="neg-phone"
   456→                                value={contactPhone}
   457→                                onChange={(e) => setContactPhone(e.target.value)}
   458→                                placeholder="+234 801 234 5678"
   459→                              />
   460→                            </div>
   461→                            <div className="grid gap-1.5">
   462→                              <Label htmlFor="neg-amount">Proposed amount (NGN)</Label>
   463→                              <Input
   464→                                id="neg-amount"
   465→                                type="number"
   466→                                value={proposedAmount}
   467→                                onChange={(e) => setProposedAmount(e.target.value)}
   468→                                placeholder={String(activation.calculatedAmount)}
   469→                                min={0}
   470→                              />
   471→                            </div>
   472→                          </div>
   473→                          <div className="grid gap-1.5">
   474→                            <Label htmlFor="neg-message">Message</Label>
   475→                            <Textarea
   476→                              id="neg-message"
   477→                              value={message}
   478→                              onChange={(e) => setMessage(e.target.value)}
   479→                              rows={4}
   480→                              placeholder="Tell us about your event, expected voter count, timeline, budget constraints…"
   481→                            />
   482→                          </div>
   483→
   484→                          {/* Preferred response channel */}
   485→                          <div className="grid gap-1.5">
   486→                            <Label>Preferred response channel</Label>
   487→                            <p className="text-[11px] text-muted-foreground">
   488→                              How should the Votewise platform admin respond to your request? You can also
   489→                              include your details below for an easy activation call.
   490→                            </p>
   491→                            <div className="grid grid-cols-3 gap-2">
   492→                              {([
   493→                            { value: "EMAIL", label: "Email", icon: "📧", desc: "Reach me by email" },
   494→                            { value: "WHATSAPP", label: "WhatsApp", icon: "💬", desc: "Chat on WhatsApp" },
   495→                            { value: "PHONE", label: "Phone call", icon: "📞", desc: "Call me directly" },
   496→                          ] as const).map((opt) => (
   497→                                <button
   498→                                  key={opt.value}
   499→                                  type="button"
   500→                                  onClick={() => setPreferredResponseChannel(opt.value)}
   501→                                  className={cn(
   502→                                    "flex flex-col items-start gap-0.5 rounded-lg border p-3 text-left transition-all",
   503→                                    preferredResponseChannel === opt.value
   504→                                      ? "border-primary bg-primary/5 ring-2 ring-primary/20"
   505→                                      : "border-border hover:border-primary/40"
   506→                                  )}
   507→                                >
   508→                                  <span className="text-base">{opt.icon}</span>
   509→                                  <span className="text-xs font-semibold">{opt.label}</span>
   510→                                  <span className="text-[10px] text-muted-foreground">{opt.desc}</span>
   511→                                </button>
   512→                              ))}
   513→                            </div>
   514→                            {preferredResponseChannel !== "EMAIL" && !contactPhone.trim() && (
   515→                              <p className="text-[11px] text-amber-600 dark:text-amber-400">
   516→                                Please include your phone number above so the platform admin can reach you via {preferredResponseChannel === "WHATSAPP" ? "WhatsApp" : "phone"}.

   521→                          <Button type="submit" size="lg" disabled={submitting}>

   529→                              </>
   530→                            )}
   531→                          </Button>
   532→                          <p className="text-center text-[11px] text-muted-foreground">
   533→                            Standard rate:{" "}
   534→                            <span className="font-medium">
   535→                              {formatCurrency(activation.calculatedAmount, activation.currency)}
   536→                            </span>{" "}
   537→                            · You can propose a lower amount and we&apos;ll respond within one business day.
   538→                          </p>
   539→                        </form>
   540→                      </TabsContent>
   541→                    </Tabs>
   542→                  )}
   543→                </CardContent>
   544→              </Card>
   545→            </div>
   546→          </>
   547→        ) : null}
   548→      </div>
   549→
   550→      <AlertDialog open={payOpen} onOpenChange={setPayOpen}>
   551→        <AlertDialogContent>
   552→          <AlertDialogHeader>

   554→              <CreditCard className="h-5 w-5 text-primary" />
   555→              Confirm payment
   556→            </AlertDialogTitle>
   557→            <AlertDialogDescription>
   558→              {activation && (
   559→                <>
   560→                  You are about to pay{" "}
   561→                  <span className="font-semibold text-foreground">
   562→                    {formatCurrency(activation.calculatedAmount, activation.currency)}
   563→                  </span>{" "}
   564→                  for {formatNumber(activation.voterCount)} eligible voters. A receipt with a verifiable
   565→                  reference will be issued instantly. The election will be marked as activated.
   566→                </>
   567→              )}
   568→            </AlertDialogDescription>
   569→          </AlertDialogHeader>
   570→          <AlertDialogFooter>
   571→            <AlertDialogCancel disabled={paying}>Cancel</AlertDialogCancel>
   572→            <AlertDialogAction
   573→              onClick={(e) => {
   574→                e.preventDefault();
   575→                pay();
   576→              }}
   577→              disabled={paying}
   578→              className="bg-primary text-primary-foreground hover:bg-primary/90"
   579→            >
   580→              {paying ? (
   581→                <>
   582→                  <Loader2 className="h-4 w-4 animate-spin" /> Processing…
   583→                </>
   584→              ) : (
   585→                <>
   586→                  <CreditCard className="h-4 w-4" /> Pay now
   587→                </>
   588→              )}
   589→            </AlertDialogAction>
   590→          </AlertDialogFooter>
   591→        </AlertDialogContent>
   592→      </AlertDialog>
   593→
   594→      {/* ─── Celebratory Activation Window ─── */}
   595→      <Dialog open={showCelebration} onOpenChange={setShowCelebration}>
   596→        <DialogContent className="sm:max-w-lg">
   597→          <motion.div
   598→            initial={{ scale: 0.8, opacity: 0 }}
   599→            animate={{ scale: 1, opacity: 1 }}
   600→            transition={{ type: "spring", duration: 0.5 }}
   601→            className="flex flex-col items-center py-6 text-center"

   609→              <CheckCircle2 className="h-10 w-10" />
   610→            </motion.div>
   611→
   612→            <h2 className="mt-4 text-2xl font-bold tracking-tight">Election Activated! 🎉</h2>

   614→              Your election is now activated and ready to go live. Share the voting link below with your voters.
   615→            </p>
   616→
   617→            {/* Subdomain link — uses the org's dedicated subdomain */}

   634→                        : `${window.location.origin}/vote/${electionId}`;

   646→                  : "Share this link with your voters."}
   647→              </p>
   648→            </div>
   649→
   650→            {/* Org homepage link */}
   651→            {orgSlug && (
   652→              <div className="mt-3 w-full">
   653→                <Label className="text-xs font-medium uppercase tracking-wide text-muted-foreground">

   655→                </Label>
   656→                <div className="mt-1.5 flex items-center gap-2 rounded-lg border bg-muted/30 p-3">

   661→                  <Button
   662→                    size="sm"
   663→                    variant="ghost"

   681→
   682→            <div className="mt-6 flex gap-3">
   683→              <Button variant="outline" onClick={() => setShowCelebration(false)}>
   684→                Close
   685→              </Button>

   687→                <a href={`/vote/${electionId}`} target="_blank" rel="noopener">
   688→                  Go to election

   690→                </a>

   692→            </div>
   693→          </motion.div>

   697→  );
