   172→function LandingInner() {
   173→  const params = useParams<{ id: string }>();
   174→  const router = useRouter();
   175→  const electionId = params.id;
   176→
   177→  const [info, setInfo] = React.useState<ElectionPublicInfo | null>(null);
   178→  const [loading, setLoading] = React.useState(true);
   179→  const [loadError, setLoadError] = React.useState<string | null>(null);
   180→
   181→  const [lookup, setLookup] = React.useState("");
   182→  const [submitting, setSubmitting] = React.useState(false);
   183→  const [error, setError] = React.useState<string | null>(null);
   184→  const [showPhotoPrompt, setShowPhotoPrompt] = React.useState(false);
   185→  const [photoVerified, setPhotoVerified] = React.useState(false);
   186→  const [pendingVoterId, setPendingVoterId] = React.useState<string | null>(null);
   187→  const [photoUploading, setPhotoUploading] = React.useState(false);
   188→  const [photoError, setPhotoError] = React.useState<string | null>(null);
   189→  const photoInputRef = React.useRef<HTMLInputElement>(null);
   190→  const [voterInfo, setVoterInfo] = React.useState<{ voterId: string; voterName: string } | null>(null);
   191→
   192→  const reduce = useReducedMotion();
   193→
   194→  React.useEffect(() => {
   195→    let cancelled = false;
   196→    async function load() {
   197→      setLoading(true);
   198→      setLoadError(null);
   199→      const res = await apiFetch<ElectionPublicInfo>(
   200→        `/api/public/results/${electionId}`
   201→      );
   202→      if (cancelled) return;
   203→      if (!res.success || !res.data) {
   204→        setLoadError(
   205→          res.error?.message ?? "We couldn't load this election."
   206→        );
   207→      } else {
   208→        setInfo(res.data);
   209→      }
   210→      setLoading(false);
   211→    }
   212→    void load();
   213→    return () => {
   214→      cancelled = true;
   215→    };
   216→  }, [electionId]);
   217→
   218→  const status = info ? (info.published ? info.election.status : info.status) : null;
   219→  const electionName = info
   220→    ? info.published
   221→      ? info.election.name
   222→      : info.electionName
   223→    : null;
   224→  const isLive = status === "LIVE";
   225→  const isPublished = info?.published === true;
   226→  const isClosed =
   227→    status === "CLOSED" ||
   228→    status === "RESULTS_REVIEW" ||
   229→    status === "ARCHIVED" ||
   230→    isPublished;
   231→  const isScheduled = status === "SCHEDULED" || status === "READY";
   232→
   233→  async function onSubmit(e: React.FormEvent) {
   234→    e.preventDefault();
   235→    if (!isLive) return;
   236→    const trimmed = lookup.trim();
   237→    if (!trimmed) {
   238→      setError("Enter your voter ID, email, phone, or matric number.");
   239→      return;
   240→    }
   241→    setError(null);
   242→    setSubmitting(true);
   243→    try {
   244→      const res = await apiFetch<VerifySendResponse>("/api/voter/verify", {
   245→        method: "POST",
   246→        body: JSON.stringify({
   247→          electionId,
   248→          voterId: trimmed,
   249→        }),
   250→      });
   251→
   252→      if (!res.success || !res.data) {
   253→        const msg = res.error?.message ?? "Something went wrong. Try again.";
   254→        setError(msg);
   255→        toast.error("Couldn't send code", { description: msg });
   256→        return;
   257→      }
   258→
   259→      const data = res.data;
   260→
   261→      // Handle already-voted
   262→      if ("alreadyVoted" in data && data.alreadyVoted) {
   263→        toast.info("Already voted", {
   264→          description: "A ballot was already cast from your account.",
   265→        });
   266→        router.push(
   267→          `/vote/${electionId}/receipt?alreadyVoted=1&voterId=${encodeURIComponent(
   268→            data.voterId
   269→          )}`
   270→        );
   271→        return;
   272→      }
   273→
   274→      if (data.sent === false) {
   275→        setError("Too many attempts. Please wait a minute and try again.");
   276→        return;
   277→      }
   278→
   279→      // ─── Device fingerprint check ───
   280→      // Check if this device has been used by another voter in this election
   281→      if (!photoVerified && data.voterId) {
   282→        const fingerprint = {
   283→          userAgent: navigator.userAgent,
   284→          screen: `${screen.width}x${screen.height}x${screen.colorDepth}`,
   285→          timezone: Intl.DateTimeFormat().resolvedOptions().timeZone,
   286→          language: navigator.language,
   287→          platform: navigator.platform,
   288→        };
   289→
   290→        const deviceRes = await apiFetch<{ requirePhoto: boolean }>("/api/voter/device-check", {
   291→          method: "POST",
   292→          body: JSON.stringify({ electionId, voterId: data.voterId, fingerprint }),
   293→        });
   294→
   295→        if (deviceRes.success && deviceRes.data?.requirePhoto) {
   296→          // Device seen before — require a live photo before proceeding.
   297→          // No reason is given to the voter, per spec.
   298→          setPendingVoterId(data.voterId);
   299→          setShowPhotoPrompt(true);
   300→          setSubmitting(false);
   301→
   302→          // Store voter info for after photo verification
   303→          try {
   304→            sessionStorage.setItem(
   305→              `votewise:voter:${electionId}`,
   306→              JSON.stringify({ voterId: data.voterId, channel: data.channel, ts: Date.now() })
   307→            );
   308→          } catch { /* ignore */ }
   309→
   310→          setVoterInfo({ voterId: data.voterId, voterName: `Voter ${data.voterId.slice(-6)}` });
   311→          // OTP has already been sent — but do NOT navigate yet. The voter
   312→          // must upload a live photo before being allowed to proceed to the
   313→          // verify page. See handlePhotoUpload() below.
   314→          return;
   315→        }
   316→      }
   317→
   318→      // Persist for cross-page recovery
   319→      try {
   320→        sessionStorage.setItem(
   321→          `votewise:voter:${electionId}`,
   322→          JSON.stringify({
   323→            voterId: data.voterId,
   324→            channel: data.channel,
   325→            ts: Date.now(),
   326→          })
   327→        );
   328→      } catch {
   329→        // sessionStorage can throw in private mode — ignore.
   330→      }
   331→
   332→      setVoterInfo({ voterId: data.voterId, voterName: `Voter ${data.voterId.slice(-6)}` });
   333→
   334→      const params = new URLSearchParams({
   335→        voterId: data.voterId,
   336→      });
   337→      if (data.channel) params.set("channel", data.channel);
   338→      router.push(`/vote/${electionId}/verify?${params.toString()}`);
   339→    } finally {
   340→      setSubmitting(false);
   341→    }
   342→  }
   343→
   344→  async function handlePhotoUpload(file: File) {
   345→    if (!pendingVoterId) return;
   346→    setPhotoUploading(true);
   347→    setPhotoError(null);
   348→    try {
   349→      const formData = new FormData();
   350→      formData.append("file", file);
   351→      formData.append("voterId", pendingVoterId);
   352→      formData.append("electionId", electionId);
   353→      formData.append("reason", "device_reuse_verification");
   354→      const res = await fetch("/api/support/upload", {
   355→        method: "POST",
   356→        body: formData,
   357→      });
   358→      const data = await res.json();
   359→      if (data.success) {
   360→        setPhotoVerified(true);
   361→        setShowPhotoPrompt(false);
   362→        toast.success("Verification complete", {
   363→          description: "You can now continue to enter your one-time code.",
   364→        });
   365→        // Now navigate to the verify page (OTP was already sent earlier)
   366→        const voterData = sessionStorage.getItem(`votewise:voter:${electionId}`);
   367→        const channel = voterData ? (JSON.parse(voterData).channel ?? undefined) : undefined;
   368→        const params = new URLSearchParams({ voterId: pendingVoterId });
   369→        if (channel) params.set("channel", channel);
   370→        router.push(`/vote/${electionId}/verify?${params.toString()}`);
   371→      } else {
   372→        setPhotoError(data.error?.message ?? "Upload failed. Please try again.");
   373→      }
   374→    } catch {
   375→      setPhotoError("Network error. Please check your connection and try again.");
   376→    } finally {
   377→      setPhotoUploading(false);
   378→    }
   379→  }
   380→
   381→  if (loading) return <LandingSkeleton />;
   382→
   383→  if (loadError || !info || !status || !electionName) {
   384→    return (
   385→      <Card className="border-destructive/30">
   386→        <CardHeader>
   387→          <CardTitle className="flex items-center gap-2 text-destructive">
   388→            <AlertCircle className="size-5" />
   389→            Election unavailable
   390→          </CardTitle>
   391→          <CardDescription>
   392→            {loadError ??
   393→              "We couldn't load this election. The link may be invalid or the election may have been archived."}
   394→          </CardDescription>
   395→        </CardHeader>
   396→        <CardContent>
   397→          <Button asChild variant="outline">
   398→            <a href="/">Back to home</a>
   399→          </Button>
   400→        </CardContent>
   401→      </Card>
   402→    );
   403→  }
   404→
   405→  return (
   406→    <div className="space-y-6">
   407→      <VoterProgress current="verify" />
   408→
   409→      {/* Election announcements from admin */}
   410→      <AnnouncementBanner electionId={electionId} />
   411→
   412→      {/* Hero — themed by org's chosen voter template */}
   413→      {(() => {
   414→        const tplId = (info.published ? info.voterTemplate : info.voterTemplate) ?? "classic";
   415→        const tpl = TEMPLATE_STYLES[tplId] ?? TEMPLATE_STYLES.classic;
   416→        return (
   417→          <motion.div
   418→            initial={reduce ? false : { opacity: 0, y: 12 }}
   419→            animate={{ opacity: 1, y: 0 }}
   420→            transition={{ duration: 0.45, ease: "easeOut" }}
   421→            className={`relative overflow-hidden rounded-2xl border bg-gradient-to-br ${tpl.heroGradient} p-6 text-center sm:p-8`}
   422→          >
   423→            <div className={`absolute right-0 top-0 h-32 w-32 -translate-y-8 translate-x-8 rounded-full ${tpl.heroRing} blur-3xl`} />
   424→            <div className="absolute bottom-0 left-1/4 h-24 w-24 translate-y-6 rounded-full bg-chart-2/10 blur-3xl" />
   425→            <div className="relative">
   426→              <div className="mb-3 flex justify-center">
   427→                <StatusBadge status={status} className="text-sm" />
   428→              </div>
   429→              <h1 className={`text-balance text-2xl font-bold leading-tight sm:text-3xl ${tpl.headingClass}`}>
   430→                {electionName}
   431→              </h1>
   432→              <p className="mx-auto mt-2 max-w-md text-balance text-sm text-muted-foreground sm:text-base">
   433→                {isPublished
   434→                  ? "This election has concluded and the results are now public."
   435→                  : isLive
   436→                    ? "Verify your identity to receive your ballot. Your vote is secret and your choices cannot be traced back to you."
   437→                    : "Welcome. Voting for this election is not currently open."}
   438→              </p>
   439→              {tplId !== "classic" && (
   440→                <div className="mt-3 inline-flex items-center gap-1 rounded-full border border-border/60 bg-background/60 px-3 py-1 text-[10px] text-muted-foreground backdrop-blur">
   441→                  <Sparkles className="size-3" /> {tplId.charAt(0).toUpperCase() + tplId.slice(1)} theme
   442→                </div>
   443→              )}
   444→            </div>
   445→          </motion.div>
   446→        );
   447→      })()}
   448→
   449→      {/* Closed / Scheduled / Published states */}
   450→      {!isLive && (
   451→        <Card className="border-border/70">
   452→          <CardHeader className="items-center text-center">
   453→            <div className="mx-auto grid size-12 place-items-center rounded-full bg-secondary">
   454→              {isClosed ? (
   455→                <CheckCircle2 className="size-6 text-primary" />
   456→              ) : (
   457→                <Clock className="size-6 text-muted-foreground" />
   458→              )}
   459→            </div>
   460→            <CardTitle className="text-xl">
   461→              {isPublished
   462→                ? "Results are public"
   463→                : isClosed
   464→                  ? "Voting has closed"
   465→                  : isScheduled
   466→                    ? "Voting opens soon"
   467→                    : "Voting is not yet open"}
   468→            </CardTitle>
   469→            <CardDescription>
   470→              {isPublished
   471→                ? "You can view the final tally for this election on the public results page."
   472→                : isClosed
   473→                  ? "Ballot casting for this election is no longer available. Results will be published after review."
   474→                  : isScheduled
   475→                    ? "Check back shortly. The election will go live at the scheduled start time."
   476→                    : "Election administrators are still preparing this election. Please check back later."}
   477→            </CardDescription>
   478→          </CardHeader>
   479→          {isPublished && (
   480→            <CardContent className="flex justify-center">
   481→              <Button asChild>
   482→                <a href={`/results/${electionId}`}>
   483→                  View results
   484→                  <ArrowRight className="size-4" />
   485→                </a>
   486→              </Button>
   487→            </CardContent>
   488→          )}
   489→        </Card>
   490→      )}
   491→
   492→      {/* Lookup card — only when LIVE */}
   493→      {isLive && (
   494→        <motion.div
   495→          initial={reduce ? false : { opacity: 0, y: 14 }}
   496→          animate={{ opacity: 1, y: 0 }}
   497→          transition={{ duration: 0.45, delay: 0.1, ease: "easeOut" }}
   498→        >
   499→          <Card className="border-border/70 shadow-sm">
   500→            <CardHeader className="gap-1.5">
   501→              <CardTitle className="text-lg">Verify your identity</CardTitle>
   502→              <CardDescription>
   503→                Enter the identifier your institution registered you with.
   504→                We&apos;ll send a one-time code to confirm it&apos;s you.
   505→              </CardDescription>
   506→            </CardHeader>
   507→            <CardContent>
   508→              {error && (
   509→                <Alert variant="destructive" className="mb-4">
   510→                  <AlertCircle className="size-4" />
   511→                  <AlertDescription>{error}</AlertDescription>
   512→                </Alert>
   513→              )}
   514→
   515→              <form onSubmit={onSubmit} className="space-y-4" noValidate>
   516→                <div className="space-y-2">
   517→                  <Label htmlFor="lookup" className="text-sm font-medium">
   518→                    Voter ID, email, phone or matric number
   519→                  </Label>
   520→                  <div className="relative">
   521→                    <IdCard className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" />
   522→                    <Input
   523→                      id="lookup"
   524→                      inputMode="text"
   525→                      autoComplete="off"
   526→                      spellCheck={false}
   527→                      placeholder="e.g. UNIZIK/2020/1000 or you@unizik.edu.ng"
   528→                      className="h-12 pl-9 text-base"
   529→                      value={lookup}
   530→                      onChange={(e) => setLookup(e.target.value)}
   531→                      disabled={submitting}
   532→                      aria-describedby="lookup-help"
   533→                    />
   534→                  </div>
   535→                  <p
   536→                    id="lookup-help"
   537→                    className="text-xs text-muted-foreground"
   538→                  >
   539→                    Use any of: your voter ID, registered email, phone number,
   540→                    or matric number.
   541→                  </p>
   542→                </div>
   543→
   544→                <Button
   545→                  type="submit"
   546→                  size="lg"
   547→                  disabled={submitting || !lookup.trim()}
   548→                  className="h-12 w-full text-base"
   549→                >
   550→                  {submitting ? (
   551→                    <>
   552→                      <Loader2 className="size-4 animate-spin" />
   553→                      Sending code…
   554→                    </>
   555→                  ) : (
   556→                    <>
   557→                      Continue
   558→                      <ChevronRight className="size-4" />
   559→                    </>
   560→                  )}
   561→                </Button>
   562→              </form>
   563→
   564→              <div className="mt-5 flex items-center justify-center gap-2 text-xs text-muted-foreground">
   565→                <LockKeyhole className="size-3.5 text-primary" />
   566→                <span>
   567→                  Encrypted in transit · We never store your code in plain
   568→                  text.
   569→                </span>
   570→              </div>
   571→            </CardContent>
   572→          </Card>
   573→        </motion.div>
   574→      )}
   575→
   576→      {/* How voting works */}
   577→      <Card className="overflow-hidden border-border/60 bg-secondary/30">
