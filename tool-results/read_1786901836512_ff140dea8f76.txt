    74→    74→
    75→    75→export default function ActivatePage({
    76→    76→  params,
    77→    77→}: {
    78→    78→  params: Promise<{ id: string }>;
    79→    79→}) {
    80→    80→  const [electionId, setElectionId] = useState<string | null>(null);
    81→    81→  const [activation, setActivation] = useState<CommercialActivationDTO | null>(null);
    82→    82→  const [loading, setLoading] = useState(true);
    83→    83→  const [error, setError] = useState<string | null>(null);
    84→    84→  const [payOpen, setPayOpen] = useState(false);
    85→    85→  const [paying, setPaying] = useState(false);
    86→    86→  const [paymentResult, setPaymentResult] = useState<{ reference: string; amount: number } | null>(null);
    87→    87→
    88→    88→  // Negotiation form state
    89→    89→  const [contactName, setContactName] = useState("");
    90→    90→  const [contactEmail, setContactEmail] = useState("");
    91→    91→  const [contactPhone, setContactPhone] = useState("");
    92→    92→  const [preferredResponseChannel, setPreferredResponseChannel] = useState<"EMAIL" | "WHATSAPP" | "PHONE">("EMAIL");
    93→    93→  const [message, setMessage] = useState("");
    94→    94→  const [proposedAmount, setProposedAmount] = useState("");
    95→    95→  const [submitting, setSubmitting] = useState(false);
    96→    96→  const [negotiationSent, setNegotiationSent] = useState(false);
    97→    97→  const [showCelebration, setShowCelebration] = useState(false);
    98→    98→  const [orgSlug, setOrgSlug] = useState<string | null>(null);
    99→    99→  const [orgName, setOrgName] = useState<string | null>(null);
   100→   100→  const [hasAutoOpened, setHasAutoOpened] = useState(false);
   101→   101→
   102→   102→  useEffect(() => {
   103→   103→    (async () => {
   104→   104→      const { id } = await params;
   105→   105→      setElectionId(id);
   106→   106→    })();
   107→   107→  }, [params]);
   108→   108→
   109→   109→  const load = useCallback(async () => {
   110→   110→    if (!electionId) return;
   111→   111→    setLoading(true);
   112→   112→    setError(null);
   113→   113→    const res = await apiFetch<ActivationResponse & { organization?: { slug: string; name: string; domain?: string | null; domainStatus?: string | null } | null }>(
   114→   114→      `/api/elections/${electionId}/activation`
   115→   115→    );
   116→   116→    setLoading(false);
   117→   117→    if (!res.success || !res.data) {
   118→   118→      setError(res.error?.message ?? "Could not load activation");
   119→   119→      return;
   120→   120→    }
   121→   121→    setActivation(res.data.activation);
   122→   122→    if (res.data.organization) {
   123→   123→      setOrgSlug(res.data.organization.slug);
   124→   124→      setOrgName(res.data.organization.name);
   125→   125→    }
   126→   126→    // Auto-open the celebration when activation has been approved
   127→   127→    // (either via payment or via negotiation approval by the platform admin)
   128→   128→    // and we haven't shown it yet for this session.
   129→   129→    const isActivated =
   130→   130→      res.data.activation.status === "MANUALLY_APPROVED" ||
   131→   131→      res.data.activation.status === "PAYMENT_VERIFIED";
   132→   132→    if (isActivated && !hasAutoOpened) {
   133→   133→      setShowCelebration(true);
   134→   134→      setHasAutoOpened(true);
   135→   135→    }
   136→   136→  }, [electionId, hasAutoOpened]);
   137→   137→
   138→   138→  useEffect(() => {
   139→   139→    let cancelled = false;
   140→   140→    (async () => {
   141→   141→      if (cancelled) return;
   142→   142→      await load();
   143→   143→    })();
   144→   144→    return () => {
   145→   145→      cancelled = true;
   146→   146→    };
   147→   147→  }, [load]);
   148→   148→
   149→   149→  async function pay() {
   150→   150→    if (!electionId) return;
   151→   151→    setPaying(true);
   152→   152→    const res = await apiFetch<PayResponse>(
   153→   153→      `/api/elections/${electionId}/activation/pay`,
   154→   154→      { method: "POST" }
   155→   155→    );
   156→   156→    setPaying(false);
   157→   157→    setPayOpen(false);
   158→   158→    if (!res.success || !res.data?.payment) {
   159→   159→      toast.error("Payment failed", { description: res.error?.message });
   160→   160→      return;
   161→   161→    }
   162→   162→    setPaymentResult({
   163→   163→      reference: res.data.payment.reference,
   164→   164→      amount: res.data.payment.amount,
   165→   165→    });
   166→   166→    toast.success("Payment received!", {
   167→   167→      description: `Reference: ${res.data.payment.reference}`,
   168→   168→    });
   169→   169→    setShowCelebration(true);
   170→   170→    load();
   171→   171→  }
   172→   172→
   173→   173→  async function submitNegotiation(e: React.FormEvent) {
   174→   174→    e.preventDefault();
   175→   175→    if (!electionId) return;
   176→   176→    if (contactName.trim().length < 2) {
   177→   177→      toast.error("Contact name is too short");
   178→   178→      return;
   179→   179→    }
   180→   180→    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(contactEmail)) {
   181→   181→      toast.error("Enter a valid email");
   182→   182→      return;
   183→   183→    }
   184→   184→    setSubmitting(true);
   185→   185→    const body = {
   186→   186→      contactName: contactName.trim(),
   187→   187→      contactEmail: contactEmail.trim(),
   188→   188→      contactPhone: contactPhone.trim() || null,
   189→   189→      preferredResponseChannel,
   190→   190→      message: message.trim() || null,
   191→   191→      proposedAmount: proposedAmount ? Number(proposedAmount) : null,
   192→   192→    };
   193→   193→    const res = await apiFetch<{ negotiation: unknown }>(
   194→   194→      `/api/elections/${electionId}/activation/negotiate`,
   195→   195→      { method: "POST", body: JSON.stringify(body) }
   196→   196→    );
   197→   197→    setSubmitting(false);
   198→   198→    if (!res.success) {
   199→   199→      toast.error("Could not submit request", { description: res.error?.message });
   200→   200→      return;
   201→   201→    }
   202→   202→    toast.success("Negotiation request sent", {
   203→   203→      description: "Our team will reach out within one business day.",
   204→   204→    });
   205→   205→    setNegotiationSent(true);
   206→   206→    load();
   207→   207→  }
   208→   208→
   209→   209→  const isVerified = activation?.status === "PAYMENT_VERIFIED" || activation?.status === "MANUALLY_APPROVED";
   210→   210→
   211→   211→  return (
   212→   212→    <ElectionShell electionId={electionId ?? ""} activeTab="activate">
   213→   213→      <div className="space-y-4">
   214→   214→        <div className="flex flex-wrap items-center justify-between gap-3">
   215→   215→          <div>
   216→   216→            <h2 className="text-lg font-semibold tracking-tight">Activation</h2>
   217→   217→            <p className="text-sm text-muted-foreground">
   218→   218→              Paid-plan elections require activation before going LIVE. Choose to pay the standard rate or request a negotiation.
   219→   219→            </p>
   220→   220→          </div>
   221→   221→          {activation && (
   222→   222→            <Badge
   223→   223→              variant="outline"
   224→   224→              className={cn("text-xs", STATUS_LABELS[activation.status]?.tone ?? STATUS_LABELS.UNPRICED.tone)}
   225→   225→            >
   226→   226→              {STATUS_LABELS[activation.status]?.label ?? activation.status.replace(/_/g, " ").toLowerCase()}
   227→   227→            </Badge>
   228→   228→          )}
   229→   229→        </div>
   230→   230→
   231→   231→        {loading ? (
   232→   232→          <div className="grid gap-4 sm:grid-cols-3">
   233→   233→            {Array.from({ length: 3 }).map((_, i) => (
   234→   234→              <div key={i} className="h-32 animate-pulse rounded-xl bg-muted" />
   235→   235→            ))}
   236→   236→          </div>
   237→   237→        ) : error ? (
   238→   238→          <ErrorState message={error} onRetry={load} />
   239→   239→        ) : activation ? (
   240→   240→          <>
   241→   241→            {/* Already activated state */}
   242→   242→            {isVerified && (
   243→   243→              <Card className="overflow-hidden border-primary/30 bg-primary/5">
   244→   244→                <CardContent className="flex flex-wrap items-center gap-3 p-4">
   245→   245→                  <div className="grid h-10 w-10 shrink-0 place-items-center rounded-full bg-emerald-100 text-emerald-600 dark:bg-emerald-950/40 dark:text-emerald-300">
   246→   246→                    <CheckCircle2 className="h-5 w-5" />
   247→   247→                  </div>
   248→   248→                  <div className="min-w-0 flex-1">
   249→   249→                    <p className="text-sm font-semibold">Election activated</p>
   250→   250→                    <p className="text-xs text-muted-foreground">
   251→   251→                      This election is cleared to go LIVE. Head back to Overview and transition to LIVE.
   252→   252→                    </p>
   253→   253→                  </div>
   254→   254→                  {activation.activatedAt && (
   255→   255→                    <Badge variant="outline" className="text-[10px]">
   256→   256→                      <ShieldCheck className="h-3 w-3" />
   257→   257→                      Activated {new Date(activation.activatedAt).toLocaleDateString()}
   258→   258→                    </Badge>
   259→   259→                  )}
   260→   260→                </CardContent>
   261→   261→              </Card>
   262→   262→            )}
   263→   263→
   264→   264→            <div className="grid gap-4 lg:grid-cols-3">
   265→   265→              {/* Pricing summary */}
   266→   266→              <Card className="lg:col-span-1">
   267→   267→                <CardHeader className="border-b pb-4">
   268→   268→                  <CardTitle className="flex items-center gap-2 text-base">
   269→   269→                    <Wallet className="h-4 w-4 text-primary" />
   270→   270→                    Pricing summary
   271→   271→                  </CardTitle>
   272→   272→                  <CardDescription>Per-voter rate · snapshot at quote time.</CardDescription>
   273→   273→                </CardHeader>
   274→   274→                <CardContent className="space-y-3 pt-4">
   275→   275→                  <Row
   276→   276→                    icon={Users}
   277→   277→                    label="Eligible voters"
   278→   278→                    value={formatNumber(activation.voterCount)}
   279→   279→                  />
   280→   280→                  <Row
   281→   281→                    icon={Tag}
   282→   282→                    label="Applicable rate"
   283→   283→                    value={`${formatCurrency(activation.applicableRate, activation.currency)} / voter`}
   284→   284→                  />
   285→   285→                  <Row
   286→   286→                    icon={Sparkles}
   287→   287→                    label="Pricing rule"
   288→   288→                    value={
   289→   289→                      <Badge variant="outline" className="text-[10px]">
   290→   290→                        {activation.pricingRule}
   291→   291→                      </Badge>
   292→   292→                    }
   293→   293→                  />
   294→   294→                  <Separator />
   295→   295→                  <Row
   296→   296→                    icon={CircleDollarSign}
   297→   297→                    label="Total amount"
   298→   298→                    value={
   299→   299→                      <span className="text-lg font-bold tabular-nums">
   300→   300→                        {formatCurrency(activation.calculatedAmount, activation.currency)}
   301→   301→                      </span>
   302→   302→                    }
   303→   303→                    highlight
   304→   304→                  />
   305→   305→                  <p className="text-[11px] text-muted-foreground">
   306→   306→                    Bulk pricing applies above 2,000 voters (₦300/voter instead of ₦400/voter).
   307→   307→                  </p>
   308→   308→                </CardContent>
   309→   309→              </Card>
   310→   310→
   311→   311→              {/* Action panel */}
   312→   312→              <Card className="lg:col-span-2">
   313→   313→                <CardHeader className="border-b pb-4">
   314→   314→                  <CardTitle className="text-base">Choose activation path</CardTitle>
   315→   315→                  <CardDescription>
   316→   316→                    {isVerified
   317→   317→                      ? "This election is already activated — no further action needed."
   318→   318→                      : "Pay now or request a custom quote. Both paths unlock LIVE."}
   319→   319→                  </CardDescription>
   320→   320→                </CardHeader>
   321→   321→                <CardContent className="pt-4">
   322→   322→                  {paymentResult ? (
   323→   323→                    <motion.div
   324→   324→                      initial={{ opacity: 0, y: 6 }}
   325→   325→                      animate={{ opacity: 1, y: 0 }}
   326→   326→                      className="flex flex-col items-center gap-3 py-6 text-center"
   327→   327→                    >
   328→   328→                      <div className="grid h-14 w-14 place-items-center rounded-full bg-emerald-100 text-emerald-600 dark:bg-emerald-950/40 dark:text-emerald-300">
   329→   329→                        <CheckCircle2 className="h-7 w-7" />
   330→   330→                      </div>
   331→   331→                      <div>
   332→   332→                        <p className="text-lg font-semibold">Payment confirmed</p>
   333→   333→                        <p className="text-sm text-muted-foreground">
   334→   334→                          Your election is now activated and ready to go LIVE.
   335→   335→                        </p>
   336→   336→                      </div>
   337→   337→                      <div className="mt-2 flex items-center gap-2 rounded-lg border bg-muted/40 px-4 py-2 text-sm">
   338→   338→                        <Receipt className="h-4 w-4 text-muted-foreground" />
   339→   339→                        <span className="font-mono text-xs">{paymentResult.reference}</span>
   340→   340→                      </div>
   341→   341→                      <Button variant="outline" size="sm" onClick={() => setPaymentResult(null)}>
   342→   342→                        Dismiss
   343→   343→                      </Button>
   344→   344→                    </motion.div>
   345→   345→                  ) : negotiationSent ? (
   346→   346→                    <motion.div
   347→   347→                      initial={{ opacity: 0, y: 6 }}
   348→   348→                      animate={{ opacity: 1, y: 0 }}
   349→   349→                      className="flex flex-col items-center gap-3 py-6 text-center"
   350→   350→                    >
   351→   351→                      <div className="grid h-14 w-14 place-items-center rounded-full bg-sky-100 text-sky-600 dark:bg-sky-950/40 dark:text-sky-300">
   352→   352→                        <Handshake className="h-7 w-7" />
   353→   353→                      </div>
   354→   354→                      <div>
   355→   355→                        <p className="text-lg font-semibold">Negotiation request sent</p>
   356→   356→                        <p className="text-sm text-muted-foreground">
   357→   357→                          Our team will reach out to <span className="font-medium">{contactEmail}</span> within one business day.
   358→   358→                        </p>
   359→   359→                      </div>
   360→   360→                      <Button
   361→   361→                        variant="outline"
   362→   362→                        size="sm"
   363→   363→                        onClick={() => {
   364→   364→                          setNegotiationSent(false);
   365→   365→                          setContactName("");
   366→   366→                          setContactEmail("");
   367→   367→                          setContactPhone("");
   368→   368→                          setMessage("");
   369→   369→                          setProposedAmount("");
   370→   370→                        }}
   371→   371→                      >
   372→   372→                        Submit another
   373→   373→                      </Button>
   374→   374→                    </motion.div>
   375→   375→                  ) : isVerified ? (
   376→   376→                    <div className="flex flex-col items-center gap-3 py-6 text-center text-sm text-muted-foreground">
   377→   377→                      <ShieldCheck className="h-10 w-10 text-primary" />
   378→   378→                      No further activation action required.
   379→   379→                    </div>
   380→   380→                  ) : (
   381→   381→                    <Tabs defaultValue="pay">
   382→   382→                      <TabsList className="grid w-full grid-cols-2">
   383→   383→                        <TabsTrigger value="pay">
   384→   384→                          <CreditCard className="h-3.5 w-3.5" />
   385→   385→                          Pay &amp; activate
   386→   386→                        </TabsTrigger>
   387→   387→                        <TabsTrigger value="negotiate">
   388→   388→                          <Handshake className="h-3.5 w-3.5" />
   389→   389→                          Request negotiation
   390→   390→                        </TabsTrigger>
   391→   391→                      </TabsList>
   392→   392→
   393→   393→                      <TabsContent value="pay" className="mt-4 space-y-4">
   394→   394→                        <div className="rounded-lg border bg-muted/30 p-4">
   395→   395→                          <p className="text-xs text-muted-foreground">
   396→   396→                            You will be charged
   397→   397→                          </p>
   398→   398→                          <p className="mt-1 text-2xl font-bold tabular-nums">
   399→   399→                            {formatCurrency(activation.calculatedAmount, activation.currency)}
   400→   400→                          </p>
   401→   401→                          <p className="mt-1 text-xs text-muted-foreground">
   402→   402→                            for {formatNumber(activation.voterCount)} eligible voters at{" "}
   403→   403→                            {formatCurrency(activation.applicableRate, activation.currency)}/voter
   404→   404→                          </p>
   405→   405→                        </div>
   406→   406→                        <Button
   407→   407→                          className="w-full"
   408→   408→                          size="lg"
   409→   409→                          onClick={() => setPayOpen(true)}
   410→   410→                          disabled={!activation || activation.voterCount === 0}
   411→   411→                        >
   412→   412→                          <CreditCard className="h-4 w-4" />
   413→   413→                          Pay {formatCurrency(activation.calculatedAmount, activation.currency)} &amp; activate
   414→   414→                        </Button>
   415→   415→                        {activation.voterCount === 0 && (
   416→   416→                          <p className="text-center text-xs text-muted-foreground">
   417→   417→                            Add eligible voters before activating — pricing is per voter.
   418→   418→                          </p>
   419→   419→                        )}
   420→   420→                        <p className="text-center text-[11px] text-muted-foreground">
   421→   421→                          <ShieldCheck className="mr-1 inline h-3 w-3" />
   422→   422→                          Payments are processed securely. A receipt with a verifiable reference is issued instantly.
   423→   423→                        </p>
   424→   424→                      </TabsContent>
   425→   425→
   426→   426→                      <TabsContent value="negotiate" className="mt-4">
   427→   427→                        <form onSubmit={submitNegotiation} className="grid gap-4">
   428→   428→                          <div className="grid gap-4 sm:grid-cols-2">
   429→   429→                            <div className="grid gap-1.5">
   430→   430→                              <Label htmlFor="neg-name">Contact name *</Label>
   431→   431→                              <Input
   432→   432→                                id="neg-name"
